cmake_minimum_required(VERSION 3.16.3)
project(final_project_aim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BUILD_VISION_APP "Build Linux camera/OpenVINO/gimbal applications" ON)
include(CTest)
find_package(Threads REQUIRED)
find_package(yaml-cpp CONFIG REQUIRED)
find_path(EIGEN3_INCLUDE_DIR Eigen/Core PATH_SUFFIXES eigen3)
find_path(FMT_INCLUDE_DIR fmt/format.h)
find_path(SPDLOG_INCLUDE_DIR spdlog/spdlog.h)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp)
foreach(dependency IN ITEMS EIGEN3_INCLUDE_DIR FMT_INCLUDE_DIR SPDLOG_INCLUDE_DIR JSON_INCLUDE_DIR)
  if(NOT ${dependency})
    message(FATAL_ERROR "Missing header dependency: ${dependency}")
  endif()
endforeach()
if(TARGET yaml-cpp::yaml-cpp)
  set(TASK2_YAML_TARGET yaml-cpp::yaml-cpp)
else()
  set(TASK2_YAML_TARGET yaml-cpp)
endif()
if(BUILD_VISION_APP)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Vision application requires Linux/MVS; use -DBUILD_VISION_APP=OFF for portable offline tests.")
  endif()
  find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d highgui imgcodecs videoio dnn)
else()
  find_package(OpenCV QUIET COMPONENTS core imgproc calib3d)
endif()
set(TASK2_OPENCV_INCLUDE_DIRS "" CACHE STRING "Real OpenCV header directories when OpenCV runtime is unavailable")
if(NOT OpenCV_FOUND AND NOT TASK2_OPENCV_INCLUDE_DIRS)
  message(FATAL_ERROR "Install libopencv-dev or provide TASK2_OPENCV_INCLUDE_DIRS for header-only offline tests.")
endif()
add_library(task2_dependencies INTERFACE)
target_include_directories(task2_dependencies INTERFACE
  ${PROJECT_SOURCE_DIR} ${EIGEN3_INCLUDE_DIR} ${FMT_INCLUDE_DIR}
  ${SPDLOG_INCLUDE_DIR} ${JSON_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS} ${TASK2_OPENCV_INCLUDE_DIRS})
target_compile_definitions(task2_dependencies INTERFACE SPDLOG_FMT_EXTERNAL FMT_HEADER_ONLY)
target_link_libraries(task2_dependencies INTERFACE ${TASK2_YAML_TARGET} Threads::Threads)

if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
if(BUILD_VISION_APP)
  # Respect OpenVINO_DIR/CMAKE_PREFIX_PATH supplied by the course environment.
  if(NOT OpenVINO_DIR AND EXISTS "/opt/intel/openvino_2024.6.0/runtime/cmake")
    set(OpenVINO_DIR "/opt/intel/openvino_2024.6.0/runtime/cmake")
  endif()
  find_package(OpenVINO REQUIRED COMPONENTS Runtime)
  add_subdirectory(tools)
  add_subdirectory(io)
  add_subdirectory(tasks/auto_aim)
  foreach(task IN ITEMS task_1 task_2 task_3)
    add_executable(${task} src/${task}.cpp)
    target_link_libraries(${task} PRIVATE auto_aim tools io task2_dependencies ${OpenCV_LIBS})
  endforeach()
endif()
